houses

- GET /houses - get all houses
- GET /houses/:id - get single house

teachers

- GET /teachers - get all teachers
- GET /teachers/:id - get single teacher
- POST /teachers - create teacher
- PUT /teachers/:id - update teacher
- DELETE /teachers/:id - delete teacher

students

- GET /students - get all students
- GET /students/:id - get single student
- POST /students - create student
- PUT /students/:id - update student
- DELETE /students/:id - delete student

courses

- GET /courses - get all courses
- GET /courses/:id - get single course med teacher og students
- POST /courses - create course
- PUT /courses/:id - update course name
- DELETE /courses/:id - delete course
- PUT /courses/:id/teacher - update course teacher
- POST /courses/:id/students - add student to course
- DELETE /courses/:id/students/:studentId - remove student from course
